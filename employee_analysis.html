    import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# Step 1: Generate sample dataset
np.random.seed(42)

departments = ["Finance", "Operations", "Marketing", "IT", "HR"]
regions = ["North America", "Europe", "Asia Pacific", "Latin America"]

data = []
for i in range(1, 101):
    emp_id = f"EMP{i:03d}"
    department = np.random.choice(departments)
    region = np.random.choice(regions)
    performance_score = round(np.random.uniform(60, 95), 2)
    years_experience = np.random.randint(1, 20)
    satisfaction_rating = round(np.random.uniform(3.0, 5.0), 1)
    data.append([emp_id, department, region, performance_score, years_experience, satisfaction_rating])

df = pd.DataFrame(data, columns=["employee_id", "department", "region", "performance_score", "years_experience", "satisfaction_rating"])

# Step 2: Frequency count of IT department
it_count = (df["department"] == "IT").sum()
print(f"Frequency count for IT department: {it_count}")

# Step 3: Histogram of departments
plt.figure(figsize=(8, 5))
sns.countplot(x="department", data=df, palette="Set2")
plt.title("Distribution of Employees Across Departments")
plt.xlabel("Department")
plt.ylabel("Count")
plt.tight_layout()

# Save the histogram as PNG
plt.savefig("department_histogram.png")

# Step 4: Save visualization and code output into HTML
html_file = "employee_analysis.html"
with open(html_file, "w") as f:
    f.write("

    f.write(f"
